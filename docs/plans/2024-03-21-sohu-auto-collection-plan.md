#### 实施代码框架
```python
def crawl_column_articles(column_url):
    """爬取专栏文章列表"""
    articles = []
    for page in range(1, 20):  # 假设最多 20 页
        url = f"{column_url}?page={page}"
        html = fetch(url)
        article_list = parse_article_list(html)
        if not article_list:
            break
        articles.extend(article_list)
        time.sleep(1)
    return articles
```
